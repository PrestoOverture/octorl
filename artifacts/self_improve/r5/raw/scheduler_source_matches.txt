830:     stable_ratio: float = 0.9,

832:     """
833:     Create a Warmup-Stable-Decay learning rate scheduler.
840:     Args:
841:         optimizer (:class:`~torch.optim.Optimizer`):
842:             The optimizer for which to schedule the learning rate.
843:         num_warmup_steps (:obj:`int`):
844:             The number of steps for the warmup phase.
845:         num_training_steps (:obj:`int`):
846:             The total number of training steps.
847:         min_lr_ratio (:obj:`float`, `optional`, defaults to 0.0):
848:             The minimum learning rate ratio w.r.t the initial learning rate.
849:         num_cycles (:obj:`float`, `optional`, defaults to 0.5):
850:             The number of waves in the cosine schedule during decay phase.
851:         last_epoch (:obj:`int`, `optional`, defaults to -1):
852:             The index of the last epoch when resuming training.
855:             Set to 0.0 to behave exactly like cosine schedule.
856: 
857:     Return:
858:         :obj:`torch.optim.lr_scheduler.LambdaLR` with the appropriate schedule.
859:     """
860:     remaining_steps = max(0, num_training_steps - num_warmup_steps)
861:     num_stable_steps = int(remaining_steps * stable_ratio)
862:     num_decay_steps = remaining_steps - num_stable_steps
863: 
864:     def lr_lambda(current_step):
865:         if current_step < num_warmup_steps:
866:             return float(current_step) / float(max(1, num_warmup_steps))
867:         if current_step < num_warmup_steps + num_stable_steps:
868:             return 1.0
869:         if current_step < num_training_steps:
870:             progress = float(current_step - num_warmup_steps - num_stable_steps) / float(max(1, num_decay_steps))
871:             value = max(0.0, 0.5 * (1.0 + math.cos(math.pi * float(num_cycles) * 2.0 * progress)))
872:             return (1.0 - min_lr_ratio) * value + min_lr_ratio
873:         return min_lr_ratio
